%% ─── 2. System Overview ──────────────────────────────────────────────────────
\begin{frame}{System Overview}
  Neural models act as \textbf{feature extractors} — all coaching scores are
  computed by \textbf{deterministic equations}, never predicted directly.

  \medskip
  \begin{columns}[T]
    \column{0.44\textwidth}
      \begin{block}{Pipeline}
        \small
        \begin{enumerate}\setlength\itemsep{2pt}
          \item \textbf{Audio Input} — 16\,kHz mono WAV
          \item \textbf{Feature Extraction} — pitch, note boundaries, phonemes
          \item \textbf{Reference Data} — MusicXML + TextGrid
          \item \textbf{Metric Computation} — predicted vs.\ reference
          \item \textbf{Scoring} — normalise $\rightarrow$ [0,\,100]
          \item \textbf{Interpretation} — rule-based feedback
        \end{enumerate}
      \end{block}

    \column{0.52\textwidth}
      \begin{block}{Three Models}
        \small
        \renewcommand{\arraystretch}{1.2}
        \begin{tabular}{@{}lp{5.2cm}@{}}
          \toprule
          Model & Role \\
          \midrule
          torchcrepe (CREPE CNN) & Frame-level F0 + voiced/unvoiced \\
          Wav2Vec2 + 3-class head & Note onset/offset times \\
          Wav2Vec2 + CTC & Phoneme boundary \& label \\
          \bottomrule
        \end{tabular}
        \medskip\par
        \textbf{Dataset:} GTSinger (English, EN-Alto-1)\\
        \textbf{Reference:} MusicXML (pitch, tempo) + TextGrid (phonemes)
      \end{block}
  \end{columns}
\end{frame}

%% ─── 3. Metrics ──────────────────────────────────────────────────────────────
\begin{frame}{Metrics --- Four Categories}
  \begin{columns}[T]
    \column{0.47\textwidth}
      \begin{block}{Pitch}
        \footnotesize
        $\delta_n$ = deviation in cents\\[4pt]
        \textbf{Acc:}\;
        $\frac{1}{N}\sum\mathbf{1}(|\delta_n|\le50)$\\[4pt]
        \textbf{MACE:}\;
        $\frac{1}{N}\sum|\delta_n|$\\[4pt]
        \textbf{RMSE:}\;
        $\sqrt{\frac{1}{N}\sum\delta_n^2}$
      \end{block}

    \column{0.06\textwidth}

    \column{0.47\textwidth}
      \begin{block}{Timing}
        \footnotesize
        $\epsilon_n = (\sh_n-s_n)\times10^3$\,ms\\[4pt]
        \textbf{Acc:}\;
        $\frac{1}{N}\sum\mathbf{1}(|\epsilon_n|\le50)$\\[4pt]
        \textbf{MAE:}\;
        $\frac{1}{N}\sum|\epsilon_n|$\\[4pt]
        \textbf{IOI-MAE:}\;
        $\frac{1}{M}\sum|\Delta\sh^{(i)}-\Delta s^{(i)}|$
      \end{block}
  \end{columns}

  \vspace{0.04\textheight}

  \begin{columns}[T]
    \column{0.47\textwidth}
      \begin{block}{Duration}
        \footnotesize
        $\dh_n = \eh_n-\sh_n$,\; $d_n=e_n-s_n$\\[4pt]
        \textbf{RelErr:}\;
        $\frac{1}{N}\sum\frac{|\dh_n-d_n|}{d_n}$\\[4pt]
        \textbf{Ratio:}\;
        $\frac{1}{N}\sum\frac{\dh_n}{d_n}$\\[4pt]
        \small scored as $|\mathrm{Ratio}-1|$
      \end{block}

    \column{0.06\textwidth}

    \column{0.47\textwidth}
      \begin{block}{Lyric}
        \footnotesize
        $\epsilon_p^{(a)} = (\ah_p-a_p)\times10^3$\,ms\\[4pt]
        \textbf{BndErr:}\;
        $\frac{1}{P}\sum|\epsilon_p^{(a)}|$\\[4pt]
        \textbf{WordAcc:}\;
        $\frac{|\text{matched}|}{|\text{ref}|}$\\[4pt]
        \textbf{Overlap:}\;
        $\frac{1}{P}\sum\mathbf{1}(\mathrm{ovlp}\ge0.5)$
      \end{block}
  \end{columns}
\end{frame}

%% ─── 4. Scoring ──────────────────────────────────────────────────────────────
\begin{frame}{Scoring --- Normalization, Categories, \& Final Score}
  \begin{columns}[T]
    \column{0.44\textwidth}
      \begin{block}{Normalization}
        \footnotesize
        \textbf{Error metrics} (lower is better) $\to$ piecewise-linear curve:
        \[
          S_\text{pw}(x) = s_i + \tfrac{s_{i+1}-s_i}{x_{i+1}-x_i}(x - x_i)
        \]
        \vspace{-4pt}
        \begin{tabular}{@{}lcc@{}}
          \toprule
          Metric & $x$ range & $s$ range \\
          \midrule
          MACE (¢)       & 0–200 & 100–0 \\
          Onset MAE (ms) & 0–200 & 100–0 \\
          RelDurErr      & 0–1.0 & 100–0 \\
          PhonBndErr (ms)& 0–120 & 100–0 \\
          \bottomrule
        \end{tabular}
        \medskip\par
        \textbf{Accuracy fractions} $\to S_\text{acc}(a)=a\times100$
      \end{block}

    \column{0.52\textwidth}
      \begin{block}{Category Weights \& Final Score}
        \footnotesize
        Confidence-weighted aggregation per category
        ($c_i = \min(1, N/3)$):
        \[
          S_\text{cat}
            = \frac{\sum_i w_i\,c_i\,s_i}{\sum_i w_i\,c_i}
        \]
        \renewcommand{\arraystretch}{1.1}
        \begin{tabular}{@{}lllll@{}}
          \toprule
          & \textbf{Pitch} & \textbf{Timing} & \textbf{Dur} & \textbf{Lyric} \\
          \midrule
          Acc / primary & 0.50 & 0.50 & — & 0.35 \\
          MAE component & 0.30 & 0.30 & 0.60 & 0.25 \\
          RMSE / Rhythm & 0.20 & 0.20 & 0.20 & 0.25 \\
          Boundary/Label& —    & —    & 0.20 & 0.15 \\
          \bottomrule
        \end{tabular}
        \medskip
        \[
          \boxed{S_\text{final}
            = 0.40\,S_\text{pitch}
            + 0.30\,S_\text{timing}
            + 0.15\,S_\text{dur}
            + 0.15\,S_\text{lyric}}
        \]
      \end{block}
  \end{columns}
\end{frame}
